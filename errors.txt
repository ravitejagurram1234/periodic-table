 -- 1. Active task population and important nullable fields
  SELECT
      id_type_tache,
      COUNT(*) AS active_tasks,
      SUM(CASE WHEN id_type_tache = 1
                AND nb_decimal IS NULL THEN 1 ELSE 0 END) AS sql_null_nb_decimal,
      SUM(CASE WHEN id_type_tache = 2
                AND id_sous_categorie IS NULL THEN 1 ELSE 0 END) AS doc_null_category,
      SUM(CASE WHEN id_type_tache = 4
                AND sql IS NULL THEN 1 ELSE 0 END) AS dynamic_null_sql,
      SUM(CASE WHEN id_type_tache = 5
                AND id_gabarit_fils IS NULL THEN 1 ELSE 0 END) AS compartment_null_child
  FROM qxp_tache
  WHERE is_actif = 1
  GROUP BY id_type_tache
  ORDER BY id_type_tache;

  -- 2. Duplicate exception block names
  SELECT
      COUNT(*) AS duplicate_keys,
      NVL(MAX(occurrences), 0) AS maximum_occurrences
  FROM (
      SELECT id_tache, nom_bloc, COUNT(*) AS occurrences
      FROM qxp_tache_exception
      GROUP BY id_tache, nom_bloc
      HAVING COUNT(*) > 1
  );

  -- 3. Duplicate template names inside one gabarit-template
  SELECT
      COUNT(*) AS duplicate_keys,
      NVL(MAX(occurrences), 0) AS maximum_occurrences
  FROM (
      SELECT agt.id_gabarit_template,
             t.nom,
             COUNT(*) AS occurrences
      FROM qxp_asso_gabarit_templates agt
      JOIN qxp_template t
        ON t.id_template = agt.id_template
      GROUP BY agt.id_gabarit_template, t.nom
      HAVING COUNT(*) > 1
  );
