Current treatment

[Fund Name][NAme of the fund][Date][Type of the report][Language]

There is a specific treatment for AMUNDI/S2G/CPR france to have another type of name via the feature in "Referentiel / Importer un fichier / Template d'import = "IT_AMUNDI_PERIMETER". We upload on it a file that contain information to be used for the naming of the report. Naming : 
RA_[Fund Name][Date][Language]_[Currency]_Code_xxx_1xxx
RA_[Fund Name][Date][Language]_[Currency]_Code_xxx_1xxx

Enhancement :
For this feature (and only this feature) the naming expected change like this :
PeriodicReport_[Fund name]UM11182[Date][Language][country][currency].pdf
                                Ex: PeriodicReport_SG-ACTIONS-ETATS-UNIS_UM11182_20250331_FRA_FR_EUR.pdf

AnnualReport_[Fund name]UM14174{}[Date]{}[Language]{}[country]_[currency].pdf

SELECT column_id, column_name, data_type, data_length, nullable
FROM user_tab_columns
WHERE table_name = 'QXP_AMUNDI_PERIMETER'
ORDER BY column_id;

SELECT cc.position, cc.column_name
FROM user_constraints c
JOIN user_cons_columns cc
  ON cc.owner = c.owner
 AND cc.constraint_name = c.constraint_name
WHERE c.table_name = 'QXP_AMUNDI_PERIMETER'
  AND c.constraint_type = 'P'
ORDER BY cc.position;

SELECT nom_template, contenu_template
FROM qxp_template_import_def
WHERE nom_template = 'IT_AMUNDI_PERIMETER';

SELECT fund_label, id_fund_bwr, decalog, fund_name,
       report_type, report_date, report_language
FROM qxp_amundi_perimeter
WHERE id_fund_bwr IN ('11182', '14174', 'UM11182', 'UM14174')
   OR UPPER(fund_label) LIKE '%ARCANCIA%'
   OR UPPER(fund_label) LIKE '%ACTIONS%ETATS%UNIS%';

SELECT id_langue_document, nom_langue, valeur_langue
FROM qxp_ref_langue_document
ORDER BY id_langue_document;

SELECT id_type_rapport, libelle
FROM qxp_ref_type_rapport
ORDER BY id_type_rapport;

SELECT column_name, data_type, data_length
FROM all_tab_columns
WHERE owner = 'OWB_DWH'
  AND table_name = 'REF_FUND'
  AND (column_name LIKE '%COUNTRY%'
    OR column_name LIKE '%DOMICIL%'
    OR column_name LIKE '%NATION%'
    OR column_name IN ('ID_FND_CODE', 'FND_SHT_DESCRIPTION', 'FND_CURRENCY'))
ORDER BY column_id;
