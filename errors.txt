 SELECT MIN(LENGTH(TRIM(sp.valeur))) AS min_length,
         MAX(LENGTH(TRIM(sp.valeur))) AS max_length,
         COUNT(CASE
             WHEN NOT REGEXP_LIKE(TRIM(sp.valeur), '^[+-]?[0-9]+$')
             THEN 1
         END) AS unusual_values
  FROM qxp_asso_suivi_parametres sp
  WHERE sp.id_parametre = 3;

  FROM qxp_asso_suivi_parametres sp
  WHERE sp.id_parametre = 2
  GROUP BY sp.valeur
  ORDER BY occurrences DESC
  FETCH FIRST 20 ROWS ONLY;

 SELECT
      COUNT(*) AS total_values,
      SUM(
          CASE
              WHEN REGEXP_LIKE(
                  TRIM(valeur),
                  '^[[:digit:]]{1,2}/[[:digit:]]{1,2}/[[:digit:]]{4} [[:digit:]]{1,2}:[[:digit:]]{1,2}:[[:digit:]]
                  {1,2}$'
              )
              THEN 1 ELSE 0
          END
      ) AS expected_shape,
      SUM(
          CASE
              WHEN valeur IS NULL
                OR NOT REGEXP_LIKE(
                    TRIM(valeur),
                    '^[[:digit:]]{1,2}/[[:digit:]]{1,2}/[[:digit:]]{4} [[:digit:]]{1,2}:[[:digit:]]{1,2}:
                    [[:digit:]]{1,2}$'
                )
              THEN 1 ELSE 0
          END
      ) AS other_shape
  FROM qxp_asso_suivi_parametres
  WHERE id_parametre = 2;
