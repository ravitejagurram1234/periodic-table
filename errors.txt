SELECT
      t.is_actif,
      COUNT(DISTINCT t.id_tache) AS task_count,
      COUNT(DISTINCT agt.id_gabarit) AS linked_gabarits,
      COUNT(DISTINCT art.id_run) AS linked_runs
  FROM qxp_tache t
  LEFT JOIN qxp_asso_gabarit_taches agt
    ON agt.id_tache = t.id_tache
  LEFT JOIN qxp_asso_run_taches art
    ON art.id_tache = t.id_tache
  WHERE t.id_type_tache = 9
  GROUP BY t.is_actif
  ORDER BY t.is_actif;
