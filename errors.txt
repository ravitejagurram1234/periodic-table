      CASE

          ELSE LENGTH(champs_vide)
      END AS value_length,
      CASE
          WHEN champs_vide IS NULL THEN '<NULL>'
          ELSE DUMP(champs_vide, 1016)
      END AS byte_dump,
      COUNT(*) AS task_count

    AND id_type_tache = 1
    AND output_data_type = 3
