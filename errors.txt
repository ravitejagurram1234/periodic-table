/* ============================================================================
   Q010 — Accepted persisted SQL values for task-133 run 505902

   Purpose:
     Run 505902 is the newest successful task-133 candidate and retains three
     SQL-storage rows. Compare these persisted, already-formatted values with a
     fresh execution of task 133's stored SQL using the same three parameters.
   ============================================================================ */

SELECT
    stored_data.id_suivi,
    stored_data.id_run,
    stored_data.date_generation,
    stored_data.data_type,
    stored_data.nom_bloc,
    stored_data.valeur_bloc,
    stored_data.description_bloc,
    stored_data.info_bloc
FROM qxp_data_storage stored_data
WHERE stored_data.id_suivi = 80593
  AND stored_data.id_run = 505902
  AND stored_data.data_type = 1
ORDER BY stored_data.nom_bloc;


WITH selected_task_runs AS (
    SELECT 133 AS id_tache, 505902 AS id_run FROM dual
    UNION ALL
    SELECT 206 AS id_tache, 399675 AS id_run FROM dual
)
SELECT
    selected_pair.id_tache,
    run_record.id_run,
    run_record.id_suivi,
    parameter_definition.id_parametre,
    parameter_definition.nom_parametre,
    parameter_definition.input_data_type,
    suivi_parameter.valeur
FROM selected_task_runs selected_pair
JOIN qxp_run run_record
  ON run_record.id_run = selected_pair.id_run
JOIN qxp_asso_run_taches run_task
  ON run_task.id_run = run_record.id_run
 AND run_task.id_tache = selected_pair.id_tache
JOIN qxp_asso_suivi_parametres suivi_parameter
  ON suivi_parameter.id_suivi = run_record.id_suivi
JOIN qxp_params_type_rapport parameter_definition
  ON parameter_definition.id_parametre = suivi_parameter.id_parametre
ORDER BY
    selected_pair.id_tache,
    parameter_definition.id_parametre;
