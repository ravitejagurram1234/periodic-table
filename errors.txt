SELECT
      agt.id_gabarit_template,
      t.id_template,
      NVL(t.nom, '<NULL>') AS nom,
      NVL(t.src_nom, '<NULL>') AS src_nom,
      NVL(t.src_nom_overflow, '<NULL>') AS src_nom_overflow,
      NVL(TO_CHAR(t.absolute), '<NULL>') AS absolute_value,
      NVL(TO_CHAR(t.absolute_repeat), '<NULL>') AS absolute_repeat,
      NVL(TO_CHAR(t.left), '<NULL>') AS left_value,
      NVL(TO_CHAR(t.top), '<NULL>') AS top_value,
      NVL(TO_CHAR(t.width), '<NULL>') AS width_value,
      NVL(TO_CHAR(t.height), '<NULL>') AS height_value,
      NVL(TO_CHAR(t.page_break), '<NULL>') AS page_break,
      NVL(TO_CHAR(t.column_break), '<NULL>') AS column_break
  FROM qxp_asso_gabarit_templates agt
  JOIN qxp_template t
    ON t.id_template = agt.id_template
  WHERE agt.id_gabarit_template = 3
    AND t.nom IN (
        'SWAP_N1_RSA',
        'SWAP_N2_RSA',
        'SWAP_N3_RSA',
        'SWAP_N4_RSA'
    )
  ORDER BY t.nom, t.id_template;
