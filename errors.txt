SELECT
      id_tache,
      '[' || champs_vide || ']' AS visible_value,
      LENGTH(champs_vide) AS value_length,
      DUMP(champs_vide, 1016) AS byte_dump
  FROM qxp_tache
  WHERE id_tache = 225;

SELECT
      t.id_tache,
      COUNT(DISTINCT agt.id_gabarit) AS all_linked_gabarits,
      COUNT(
          DISTINCT CASE
              WHEN g.is_actif = 1 THEN g.id_gabarit
          END
      ) AS active_linked_gabarits,
      COUNT(
          DISTINCT CASE
              WHEN g.is_actif = 1
               AND afg.id_gabarit IS NOT NULL
              THEN g.id_gabarit
          END
  FROM qxp_tache t
  LEFT JOIN qxp_asso_gabarit_taches agt
    ON agt.id_tache = t.id_tache
  LEFT JOIN qxp_gabarit g
    ON g.id_gabarit = agt.id_gabarit
  LEFT JOIN qxp_asso_fond_gabarit afg
    ON afg.id_gabarit = g.id_gabarit
  WHERE t.is_actif = 1
    AND t.id_type_tache = 1
    AND NOT EXISTS (
        SELECT 1
        FROM qxp_asso_run_taches art
        WHERE art.id_tache = t.id_tache
    )
  GROUP BY
      t.id_tache,
      t.commentaire
  ORDER BY t.id_tache;
