SELECT
      r.id_run,
      r.id_suivi,
      r.id_statut_generation,
      r.date_creation_run,
      r.date_debut_generation,
      r.date_fin_generation,
      r.id_doc_qxp,
      r.id_doc_pdf,
      s.id_gabarit,
      s.id_type_rapport,
      s.id_fnd_code,
      s.id_unit_code,
      s.date_echeance
  FROM qxp_asso_run_taches art
  JOIN qxp_run r
    ON r.id_run = art.id_run
  JOIN qxp_suivi s
    ON s.id_suivi = r.id_suivi
  WHERE art.id_tache = 53
  ORDER BY
      NVL(
          r.date_fin_generation,
          NVL(r.date_debut_generation, r.date_creation_run)
      ) DESC NULLS LAST
  FETCH FIRST 10 ROWS ONLY;

      sp.valeur
  FROM qxp_run r
  JOIN qxp_asso_suivi_parametres sp
    ON sp.id_suivi = r.id_suivi
  JOIN qxp_params_type_rapport p
    ON p.id_parametre = sp.id_parametre
  WHERE r.id_run = <CHOSEN_RUN_ID>
  ORDER BY p.id_parametre;
