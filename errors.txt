/* ============================================================================
   Q012 — Text-safe task-133 storage evidence

   Purpose:
     Excel converted DATE.N0 text and DATE_GENERATION while data was pasted.
     Prefix/wrap the VARCHAR2 fields and explicitly format the Oracle DATE so
     the next export preserves exact text, punctuation and whitespace.

   Expected result:
     Three rows. Copy/export as text without removing the TEXT:[...] wrapper.
   ============================================================================ */

SELECT
    stored_data.id_suivi,
    stored_data.id_run,
    'GENERATION:' || TO_CHAR(
        stored_data.date_generation,
        'YYYY-MM-DD HH24:MI:SS'
    ) AS date_generation_text,
    stored_data.data_type,
    stored_data.nom_bloc,
    CASE
        WHEN stored_data.valeur_bloc IS NULL THEN '<NULL>'
        ELSE 'TEXT:[' ||
             REPLACE(
                 REPLACE(stored_data.valeur_bloc, CHR(13), '<CR>'),
                 CHR(10), '<LF>'
             ) || ']'
    END AS valeur_bloc_exact,
    CASE
        WHEN stored_data.valeur_bloc IS NULL THEN NULL
        ELSE LENGTH(stored_data.valeur_bloc)
    END AS valeur_length,
    CASE
        WHEN stored_data.valeur_bloc IS NULL THEN '<NULL>'
        ELSE DUMP(stored_data.valeur_bloc, 1016)
    END AS valeur_dump,
    CASE
        WHEN stored_data.description_bloc IS NULL THEN '<NULL>'
        ELSE 'TEXT:[' || stored_data.description_bloc || ']'
    END AS description_bloc_exact,
    CASE
        WHEN stored_data.info_bloc IS NULL THEN '<NULL>'
        ELSE 'TEXT:[' || stored_data.info_bloc || ']'
    END AS info_bloc_exact
FROM qxp_data_storage stored_data
WHERE stored_data.id_suivi = 80593
  AND stored_data.id_run = 505902
  AND stored_data.data_type = 1
ORDER BY stored_data.nom_bloc;


/* ============================================================================
   Q013 — Recent task-206 run with nonzero percentage data

   Purpose:
     Run 399675 proves NULL handling only. Find a successful task-206 run whose
     exact task expression produces at least one nonzero value, so the
     NLS_NUMERIC_CHARACTERS-dependent FM990D00 output can be verified in Java.

   Next step:
     Choose one returned ID_RUN, fetch its parameters with the SQL handbook
     section 1.4, and execute task 206 in the same Oracle session.
   ============================================================================ */

WITH candidate_runs AS (
    SELECT DISTINCT
        run_record.id_run,
        run_record.id_suivi,
        run_record.date_fin_generation,
        suivi_record.id_fnd_code,
        suivi_record.id_unit_code,
        suivi_record.date_echeance
    FROM qxp_asso_run_taches run_task
    JOIN qxp_run run_record
      ON run_record.id_run = run_task.id_run
    JOIN qxp_suivi suivi_record
      ON suivi_record.id_suivi = run_record.id_suivi
    JOIN owb_dwh.ref_unit unit_reference
      ON unit_reference.id_unit_fund_code = suivi_record.id_fnd_code
     AND unit_reference.id_unit_code = suivi_record.id_unit_code
     AND unit_reference.unit_end_validity = DATE '2199-12-31'
    WHERE run_task.id_tache = 206
      AND run_record.id_statut_generation = 2
)
SELECT
    candidate_run.id_run,
    candidate_run.id_suivi,
    candidate_run.date_fin_generation,
    candidate_run.id_fnd_code AS p_code_port,
    candidate_run.date_echeance AS p_date_echeance,
    candidate_run.id_unit_code AS p_id_unit_code,
    kii_storage.tx_ffg
        + kii_storage.tx_frais_pret
        + kii_storage.tx_com_mvmt
        + kii_storage.tx_ct_ind
        + kii_storage.tx_retro AS frais_courant,
    kii_storage.tx_com_surperf AS frais_performance
FROM candidate_runs candidate_run
JOIN qxp.qxp_kii_storage kii_storage
  ON kii_storage.id_fnd_code = candidate_run.id_fnd_code
 AND kii_storage.id_unit_code = candidate_run.id_unit_code
 AND kii_storage.nav_date = candidate_run.date_echeance
 AND kii_storage.id_kii = (
        SELECT MAX(candidate_storage.id_kii)
        FROM qxp.qxp_kii_storage candidate_storage
        WHERE candidate_storage.id_fnd_code = candidate_run.id_fnd_code
          AND candidate_storage.id_unit_code = candidate_run.id_unit_code
          AND candidate_storage.nav_date = candidate_run.date_echeance
    )
WHERE (
        kii_storage.tx_ffg
        + kii_storage.tx_frais_pret
        + kii_storage.tx_com_mvmt
        + kii_storage.tx_ct_ind
        + kii_storage.tx_retro
      ) <> 0
   OR kii_storage.tx_com_surperf <> 0
ORDER BY candidate_run.date_fin_generation DESC NULLS LAST
FETCH FIRST 10 ROWS ONLY;
