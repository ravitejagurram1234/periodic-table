SELECT
      CASE
          WHEN champs_vide IS NULL THEN '<NULL>'
          ELSE '[' ||
               REPLACE(
                   REPLACE(champs_vide, CHR(13), '<CR>'),
                   CHR(10), '<LF>'
               ) ||
               ']'
      END AS champs_vide_display,
      CASE
          WHEN champs_vide IS NULL THEN NULL
          ELSE LENGTH(champs_vide)
      END AS value_length,
      CASE
          WHEN champs_vide IS NULL THEN '<NULL>'
          ELSE DUMP(champs_vide, 1016)
      END AS byte_dump,
      COUNT(*) AS task_count
  FROM qxp_tache
  WHERE is_actif = 1
    AND id_type_tache = 1
    AND output_data_type = 3
  GROUP BY champs_vide
  ORDER BY task_count DESC;

      COUNT(*) AS exception_rule_count

    ON te.id_tache = t.id_tache
