SELECT
      r.id_run,
      r.id_suivi,
      p.id_parametre,
      p.nom_parametre,
      p.input_data_type,
      sp.valeur
  FROM qxp_run r

  JOIN qxp_asso_suivi_parametres sp
    ON sp.id_suivi = r.id_suivi

  WHERE r.id_run IN (
      509474,
      464995,
      275985,
      514731
  )

  ORDER BY
      r.id_run,
      p.id_parametre;
