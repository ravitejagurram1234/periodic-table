## What was wrong in the current Java code

Before Wave 5, Java could observe or produce behavior different from the accepted engine:

- groups were evaluated after the whole project, allowing an invalid forward reference into a later spread;
- null project nodes, group children and MOVE/table/page structures were skipped where `.NET` fails;
- box type parsing, malformed geometry and decimal range/subtraction did not follow legacy conversion rules;
- modifier table defaults, whitespace presence, spread ordering and dummy-page conversion differed;
- prepare/reset/process/post-process failures could expose raw exception payloads and lose exact phase context;
- legacy six-field DID, sentinels, raw text, date parsing and generation order were not preserved;
- DID pagination scanning stopped too early and MOVE coordinates ignored the configured decimal separator;
- modifier XML lacked the accepted declaration/indentation contract and silently accepted populated `SYMMVERTEX`;
- null project serialization incorrectly produced declaration-only XML instead of failing;
- value-bearing system debug logs could expose document content.

package com.socgen.sgs.api.quark.engine.domain.element;

import com.socgen.sgs.api.quark.engine.domain.helper.CheckedDecimalArithmetic;
import com.socgen.sgs.api.quark.engine.domain.helper.InvariantValueConverter;
import com.socgen.sgs.api.quark.engine.domain.project.QxpProject;
import com.socgen.sgs.api.quark.engine.enums.TBoxTypeEnum;
import com.socgen.sgs.api.quark.engine.integration.soap.generated.Box;
import lombok.Getter;
import lombok.Setter;

/**
 * Represents a template box element available for dynamic tasks.
 * Wraps a SOAP-generated {@link Box} with geometry and type metadata.
 *
 * <p>Never work directly on a TBox without cloning it first (via TElementHelper.clone()),
 * as modifications would affect all tasks sharing the same template.
 *
 * Cross-reference: QXP.Engine.Core.TBox
 */
@Getter
@Setter
public class TBox extends TElement {

    private static final String UNKNOWN_PAGE = "?";

    /** Source SOAP box (the main box definition). */
    private Box srcBox;

    /** Extra SOAP box (used for certain operations like PDF offset). */
    private Box srcExtraBox;

    /** Box content type (CT_TEXT, CT_PICT, CT_NONE, CT_UNKNOWN). */
    private TBoxTypeEnum type = TBoxTypeEnum.CT_UNKNOWN;

    /** Page containing this element ("?" if not evaluated). */
    private String page = UNKNOWN_PAGE;

    /**
     * No-arg constructor for serialization/deserialization.
     */
    public TBox() {
        super();
    }

    /**
     * Create a TBox from a SOAP Box.
     * Extracts name, type, and page from the box properties.
     *
     * @param box the SOAP-generated Box
     */
    public TBox(Box box) {
        super(box.getName());
        this.srcBox = box;

        // Extract box type
        if (box.getBoxType() != null && !box.getBoxType().isEmpty()) {
            this.type = parseAcceptedBoxType(box.getBoxType());
        }

        // Extract page from geometry
        if (box.getGeometry() != null) {
            this.page = box.getGeometry().getPage();
        }
    }

    /**
     * Create a TBox from a SOAP Box with an extra box.
     * The extra box enables certain operations (e.g., PDF image offset).
     *
     * @param box the main SOAP-generated Box
     * @param srcExtraBox the extra SOAP-generated Box
     */
    public TBox(Box box, Box srcExtraBox) {
        this(box);
        this.srcExtraBox = srcExtraBox;
    }

    /**
     * Evaluate geometry (left, top, right, bottom, width, height) from the source box.
     *
     * @param qxpProject the QXP project (not used directly for TBox — geometry comes from srcBox)
     */
    @Override
    public void evaluate(QxpProject qxpProject) {
        if (isEvaluated()) {
            return;
        }

        String leftStr = srcBox.getGeometry().getPosition().getLeft();
        String topStr = srcBox.getGeometry().getPosition().getTop();
        String rightStr = srcBox.getGeometry().getPosition().getRight();
        String bottomStr = srcBox.getGeometry().getPosition().getBottom();

        setLeft(InvariantValueConverter.toDecimal(leftStr));
        setTop(InvariantValueConverter.toDecimal(topStr));
        setRight(InvariantValueConverter.toDecimal(rightStr));
        setBottom(InvariantValueConverter.toDecimal(bottomStr));

        setWidth(CheckedDecimalArithmetic.subtract(getRight(), getLeft()));
        setHeight(CheckedDecimalArithmetic.subtract(getBottom(), getTop()));

        setEvaluated(true);
    }

    /** Reproduces the case-sensitive, surrounding-whitespace-tolerant Enum.Parse used by the accepted engine. */
    private static TBoxTypeEnum parseAcceptedBoxType(String value) {
        String candidate = value.strip();
        if (candidate.matches("[+-]?\\d+")) {
            int numericValue = Integer.parseInt(candidate);
            TBoxTypeEnum[] values = TBoxTypeEnum.values();
            return numericValue >= 0 && numericValue < values.length
                    ? values[numericValue]
                    : TBoxTypeEnum.CT_UNKNOWN;
        }
        return TBoxTypeEnum.valueOf(candidate);
    }
}

/**
 * Represents a template group element (multiple boxes) for dynamic tasks.
 * Wraps a SOAP-generated {@link Group} and contains child TBox elements.
 *
 * <p>Never work directly on a TGroup without cloning it first,
 * as modifications would affect all tasks sharing the same template.
 *
 * Cross-reference: QXP.Engine.Core.TGroup
 */
@Getter
@Setter
public class TGroup extends TElement {

    private static final String UNKNOWN_PAGE = "?";

    /** Source SOAP group. */
    private Group srcGroup;

    /** Source boxes extracted from the group (populated after evaluation). */
    private Box[] srcBoxes;

    /** Child TBox elements contained in this group. */
    private List<TBox> tBoxes = new ArrayList<>();

    /** Page containing this element ("?" if not evaluated). */
    private String page = UNKNOWN_PAGE;

    /**
     * No-arg constructor for serialization/deserialization.
     */
    public TGroup() {
        super();
    }

    /**
     * Create a TGroup from a SOAP Group.
     *
     * @param group the SOAP-generated Group
     */
    public TGroup(Group group) {
        super(group.getName());
        this.srcGroup = group;
    }

    /**
     * Evaluate geometry by computing the bounding box of all child elements.
     * Iterates through boxRefs, resolves them from the QXP project elements,
     * and computes min/max bounds.
     *
     * @param qxpProject the QXP project containing the named elements
     */
    @Override
    public void evaluate(QxpProject qxpProject) {
        if (isEvaluated()) {
            return;
        }

        evaluateBoxes(qxpProject);

        // Compute bounding box from all child TBoxes
        if (tBoxes.isEmpty()) {
            setLeft(BigDecimal.ZERO);
            setTop(BigDecimal.ZERO);
            setWidth(BigDecimal.ZERO);
            setHeight(BigDecimal.ZERO);
        } else {
            BigDecimal leftMin = InvariantValueConverter.DECIMAL_UNSET.negate();
            BigDecimal topMin = InvariantValueConverter.DECIMAL_UNSET.negate();
            BigDecimal rightMax = InvariantValueConverter.DECIMAL_UNSET;
            BigDecimal bottomMax = InvariantValueConverter.DECIMAL_UNSET;

            for (TBox tBox : tBoxes) {
                if (tBox.getLeft().compareTo(leftMin) < 0) leftMin = tBox.getLeft();
                if (tBox.getTop().compareTo(topMin) < 0) topMin = tBox.getTop();
                if (tBox.getRight().compareTo(rightMax) > 0) rightMax = tBox.getRight();
                if (tBox.getBottom().compareTo(bottomMax) > 0) bottomMax = tBox.getBottom();
            }

            setLeft(leftMin);
            setTop(topMin);
            setWidth(CheckedDecimalArithmetic.subtract(rightMax, leftMin));
            setHeight(CheckedDecimalArithmetic.subtract(bottomMax, topMin));
        }

        setEvaluated(true);
    }

    /**
     * Evaluate child boxes by resolving boxRefs from the elements map.
     * Each boxRef name is looked up in the project elements dictionary.
     *
     * @param qxpProject project containing the global named-element registry
     */
    private void evaluateBoxes(QxpProject qxpProject) {
        boolean subGroup = false;
        List<Box> resolvedBoxes = new ArrayList<>();

        if (srcGroup.getBoxRefs() == null || srcGroup.getBoxRefs().length == 0) {
            this.srcBoxes = new Box[0];
            return;
        }

        for (BoxRef boxRef : srcGroup.getBoxRefs()) {
            String refName = boxRef.getName();

            if (!qxpProject.getElements().containsKey(refName)) {
                // Parity: .NET TGroup.cs throws Exception_TElement(TGroupUnKnownSubGroup, boxRef.name,
                // srcGroup.name); QxpProject.evaluatePendingGroups records it as an UNSPECIFIED run error.
                // (Finding #18/#19.)
                throw new ExceptionTElement(String.format(
                        "Element '%s' was not found in group '%s'",
                        refName, srcGroup.getName()));
            }

            TElement refElement = qxpProject.getElements().get(refName);

            if (refElement instanceof TBox) {
                TBox tBox = (TBox) refElement;
                tBoxes.add(tBox);
                tBox.evaluate(qxpProject);
                resolvedBoxes.add(tBox.getSrcBox());

                if (UNKNOWN_PAGE.equals(page) && !UNKNOWN_PAGE.equals(tBox.getPage())) {
                    this.page = tBox.getPage();
                }
            } else if (refElement instanceof TGroup) {
                // Sub-groups: evaluate but flag as problematic
                // .NET logs warning: "TGroupSubGroupProblem"
                subGroup = true;
                TGroup subGrp = (TGroup) refElement;
                subGrp.evaluate(qxpProject);

                if (UNKNOWN_PAGE.equals(page) && !UNKNOWN_PAGE.equals(subGrp.getPage())) {
                    this.page = subGrp.getPage();
                }
            }
        }

        this.srcBoxes = resolvedBoxes.toArray(new Box[0]);

        if (subGroup) {
            // Parity: .NET TGroup.cs throws Exception_TElement(TGroupSubGroupProblem, srcGroup.name, Page)
            // — a template element of type group must not contain other groups. Recorded as an
            // UNSPECIFIED run error by QxpProject.evaluatePendingGroups. (Finding #44/#19.)
            throw new ExceptionTElement(String.format(
                    "Group '%s' on page '%s' contains subgroups that cannot be evaluated safely",
                    srcGroup.getName(), page));
        }
    }
}

package com.socgen.sgs.api.quark.engine.domain.helper;

import java.math.BigDecimal;
import java.math.BigInteger;

/**
 * Focused emulation of the checked operations used by {@code System.Decimal} in the accepted
 * engine's box-budget calculation.
 *
 * <p>A .NET decimal stores a sign, a 96-bit integer coefficient and a scale from 0 to 28. Division
 * rounds to nearest-even when the exact quotient cannot fit that representation. Keeping this
 * behavior isolated avoids silently changing one-box inclusion decisions through an arbitrary
 * {@link java.math.MathContext} or {@link BigDecimal} scale.
 */
public final class CheckedDecimalArithmetic {

    private static final int MAX_SCALE = 28;
    private static final BigInteger MAX_COEFFICIENT = BigInteger.ONE.shiftLeft(96).subtract(BigInteger.ONE);
    private static final BigInteger INT32_MIN = BigInteger.valueOf(Integer.MIN_VALUE);
    private static final BigInteger INT32_MAX = BigInteger.valueOf(Integer.MAX_VALUE);

    private CheckedDecimalArithmetic() {
    }

    /** Subtract two values with the checked range semantics of {@code System.Decimal}. */
    public static BigDecimal subtract(BigDecimal minuend, BigDecimal subtrahend) {
        BigDecimal left = requireRepresentable(minuend);
        BigDecimal right = requireRepresentable(subtrahend);
        return requireRepresentable(left.subtract(right));
    }

    /** Divide two values using the representable range and nearest-even rounding of .NET decimal. */
    public static BigDecimal divide(BigDecimal dividend, BigDecimal divisor) {
        DecimalParts left = DecimalParts.from(dividend);
        DecimalParts right = DecimalParts.from(divisor);
        if (right.coefficient.signum() == 0) {
            throw new ArithmeticException("Attempted to divide by zero");
        }
        if (left.coefficient.signum() == 0) {
            return BigDecimal.ZERO;
        }

        boolean negative = left.negative ^ right.negative;
        BigInteger numerator = left.coefficient.multiply(BigInteger.TEN.pow(right.scale));
        BigInteger denominator = right.coefficient.multiply(BigInteger.TEN.pow(left.scale));
        BigInteger gcd = numerator.gcd(denominator);
        numerator = numerator.divide(gcd);
        denominator = denominator.divide(gcd);

        for (int scale = MAX_SCALE; scale >= 0; scale--) {
            BigInteger scaledNumerator = numerator.multiply(BigInteger.TEN.pow(scale));
            BigInteger[] quotientAndRemainder = scaledNumerator.divideAndRemainder(denominator);
            BigInteger coefficient = roundHalfEven(
                    quotientAndRemainder[0], quotientAndRemainder[1], denominator);
            if (coefficient.compareTo(MAX_COEFFICIENT) <= 0) {
                if (negative) {
                    coefficient = coefficient.negate();
                }
                return new BigDecimal(coefficient, scale);
            }
        }

        throw new ArithmeticException("Value was either too large or too small for a Decimal");
    }

    /** Checked .NET-style explicit conversion to Int32: truncate toward zero, then range-check. */
    public static int toInt32Truncate(BigDecimal value) {
        BigInteger truncated = value.toBigInteger();
        if (truncated.compareTo(INT32_MIN) < 0 || truncated.compareTo(INT32_MAX) > 0) {
            throw new ArithmeticException("Value was either too large or too small for an Int32");
        }
        return truncated.intValue();
    }

    /** Require a value to fit the 96-bit coefficient and 0..28 scale decimal representation. */
    public static BigDecimal requireRepresentable(BigDecimal value) {
        DecimalParts parts = DecimalParts.from(value);
        BigInteger coefficient = parts.negative ? parts.coefficient.negate() : parts.coefficient;
        return new BigDecimal(coefficient, parts.scale);
    }

    /**
     * Convert to the closest representable checked decimal, using nearest-even rounding when scale
     * or coefficient reduction is required. This matches checked decimal parsing/conversion rather
     * than silently accepting Java's unbounded precision.
     */
    public static BigDecimal roundToRepresentable(BigDecimal value) {
        if (value == null) throw new NullPointerException("value");
        if (value.signum() == 0) return BigDecimal.ZERO;

        BigDecimal normalized = value.stripTrailingZeros();
        BigInteger coefficient = normalized.unscaledValue().abs();
        int scale = normalized.scale();
        if (scale < 0) {
            coefficient = coefficient.multiply(BigInteger.TEN.pow(-scale));
            scale = 0;
        }

        int minimumDrop = Math.max(0, scale - MAX_SCALE);
        BigInteger representableCoefficient = null;
        int representableScale = 0;
        for (int drop = minimumDrop; drop <= scale; drop++) {
            BigInteger divisor = BigInteger.TEN.pow(drop);
            BigInteger[] quotientAndRemainder = coefficient.divideAndRemainder(divisor);
            BigInteger rounded = roundHalfEven(
                    quotientAndRemainder[0], quotientAndRemainder[1], divisor);
            if (rounded.compareTo(MAX_COEFFICIENT) <= 0) {
                representableCoefficient = rounded;
                representableScale = scale - drop;
                break;
            }
        }

        if (representableCoefficient == null) {
            throw new ArithmeticException("Value was either too large or too small for a Decimal");
        }
        if (normalized.signum() < 0) representableCoefficient = representableCoefficient.negate();
        return new BigDecimal(representableCoefficient, representableScale);
    }

    private static BigInteger roundHalfEven(BigInteger quotient, BigInteger remainder,
                                            BigInteger denominator) {
        int halfComparison = remainder.shiftLeft(1).compareTo(denominator);
        if (halfComparison > 0 || (halfComparison == 0 && quotient.testBit(0))) {
            return quotient.add(BigInteger.ONE);
        }
        return quotient;
    }

    private static final class DecimalParts {
        private final BigInteger coefficient;
        private final int scale;
        private final boolean negative;

        private DecimalParts(BigInteger coefficient, int scale, boolean negative) {
            this.coefficient = coefficient;
            this.scale = scale;
            this.negative = negative;
        }

        private static DecimalParts from(BigDecimal value) {
            if (value == null) {
                throw new NullPointerException("value");
            }
            if (value.signum() == 0) {
                return new DecimalParts(BigInteger.ZERO, 0, false);
            }

            BigDecimal normalized = value.stripTrailingZeros();
            BigInteger coefficient = normalized.unscaledValue().abs();
            int scale = normalized.scale();
            if (scale < 0) {
                coefficient = coefficient.multiply(BigInteger.TEN.pow(-scale));
                scale = 0;
            }
            if (scale > MAX_SCALE || coefficient.compareTo(MAX_COEFFICIENT) > 0) {
                throw new ArithmeticException("Value was either too large or too small for a Decimal");
            }
            return new DecimalParts(coefficient, scale, normalized.signum() < 0);
        }
    }

     *
     * @param project the SOAP-generated Project

        return project == null || project.getLayouts() == null || project.getLayouts().length == 0;

    /**
     * Analyse the project structure to build the elements dictionary.
     * Iterates through all layouts → spreads → boxes/tables/groups,
     * creating TBox/TTable/TGroup instances and adding them to the elements map.
     *
     * <p>This method is idempotent — calling it multiple times has no effect
     * after the first successful analysis.
     *
     * @param task     a reference to the calling task (for error logging to run)
     * @param logError if true, structural errors are logged to the run's error list;
     *                 if false, errors are silently ignored
     */
    public void analyse(TaskBase task, boolean logError) {
        // Don't re-analyse if already done
        if (analysed) {
            return;
        }

        elements = new LinkedHashMap<>();
        if (project.getLayouts() == null) {
            analysed = true;
            return;
        }

        for (Layout layout : project.getLayouts()) {
            for (Spread spread : layout.getSpreads()) {
                // Process boxes
                processBoxes(spread);

                // Process tables
                processTables(spread);

                // Groups can contain only elements visible by the end of their own spread.
                List<TGroup> pendingGroups = new ArrayList<>();
                processGroups(spread, pendingGroups);
                evaluatePendingGroups(pendingGroups, task, logError);
            }
        }

        analysed = true;
    }

    /**
     * Process all boxes in a spread.
     * Creates TBox instances and adds them to the elements dictionary.
     */
    private void processBoxes(Spread spread) {
        if (spread.getBoxes() == null) {
            return;
        }

        for (Box box : spread.getBoxes()) {
            TBox tBox = new TBox(box);

            // Parity: .NET QXP_Project adds to a Dictionary which throws ArgumentNullException on a null
            // key, aborting Analyse. (Finding #81.)
            if (tBox.getName() == null) {
                throw new IllegalArgumentException("Element name is null");
            }

            // Some documents have duplicate box names (legacy QXP Server 7 bug)
            if (!elements.containsKey(tBox.getName())) {
                elements.put(tBox.getName(), tBox);
                tBox.evaluate(this);
            } else {
                log.debug("Duplicate box name [{}] — skipping", tBox.getName());
            }
        }
    }

    /**
     * Process all tables in a spread.
     * Creates TTable instances and adds them to the elements dictionary.
     */
    private void processTables(Spread spread) {
        if (spread.getTables() == null) {
            return;
        }

        for (Table table : spread.getTables()) {
            // QXP Server v9 bug: there's always a null table even when none exist
            if (table == null) {
                continue;
            }

            TTable tTable = new TTable(table);

            if (tTable.getName() == null) {
                throw new IllegalArgumentException("Element name is null"); // .NET ArgumentNullException (#81)
            }

            if (!elements.containsKey(tTable.getName())) {
                elements.put(tTable.getName(), tTable);
                tTable.evaluate(this);
            } else {
                log.debug("Duplicate table name [{}] — skipping", tTable.getName());
            }
        }
    }

    /**
     * Process all groups in a spread.
     * Creates TGroup instances, adds them to the elements dictionary,
     * and collects them for deferred evaluation.
     *
     * <p>Groups are evaluated after all boxes and tables are registered
     * because groups reference child elements via boxRefs.
     */
    private void processGroups(Spread spread, List<TGroup> pendingGroups) {
        if (spread.getGroups() == null) {
            return;
        }

        for (Group group : spread.getGroups()) {
            // QXP Server v9 bug: there's always a null group even when none exist
            if (group == null) {
                continue;
            }

            TGroup tGroup = new TGroup(group);

            if (tGroup.getName() == null) {
                throw new IllegalArgumentException("Element name is null"); // .NET ArgumentNullException (#81)
            }

            if (!elements.containsKey(tGroup.getName())) {
                elements.put(tGroup.getName(), tGroup);
                pendingGroups.add(tGroup);
            } else {
                log.debug("Duplicate group name [{}] — skipping", tGroup.getName());
            }
        }
    }

    /**
     * Evaluate all pending groups.
     * Called after all boxes/tables/groups are registered in the elements map.
     *
     * @param pendingGroups the groups to evaluate
     * @param task          the calling task (for error logging)
     * @param logError      whether to log structural errors to the run
     */
    private void evaluatePendingGroups(List<TGroup> pendingGroups, TaskBase task, boolean logError) {
        for (TGroup tGroup : pendingGroups) {
            try {
                tGroup.evaluate(this);
            } catch (ExceptionTElement e) {
                // Parity: .NET QXP_Project catches ONLY Exception_TElement and, when logError, records it
                // via task.Run.Errors.Add(tex.ToString()) — the string overload → Error_Type.Unspecified (1),
                // NOT Critique. Any non-TElement exception propagates and aborts analyse, as in .NET.
                // (Findings #18/#19/#44.)
                if (logError && task != null) {
                    log.warn("Unable to evaluate group [{}] for task [{}]", tGroup.getName(), task.getId());
                    task.getRun().getErrors().add(new RunError(RunError.UNSPECIFIED,
                            "Unable to evaluate group " + tGroup.getName() + ": " + e.getMessage()));
                } else {
                    log.debug("Unable to evaluate group [{}]", tGroup.getName());
                }
            }
        }
    }

     * Prepare phase — call prepare() on EVERY task (not just todo), before processing.
     * Per-task failures are recorded as Critique errors and flag the task in error.
     * Cross-reference: .NET Run_Base.Prepare() (iterates _tasks.Values).

            try {
                log.debug("Preparing task {}", task.getId());

                        "Error while preparing task " + task.getId() + " for " + run.getDebugInfo());
            } catch (EngineException controlled) {

                run.getErrors().add(new RunError(CRITIQUE, controlled.getMessage()));
                log.error("Task phase failed: phase [prepare], task [{}], run [{}], cause [{}]",
                        task.getId(), run.getId(), causeType(controlled));
            }
        }

        log.info("Task preparation completed for runId: {}", run.getId());
    }

    @Override
    public void processTasks(Run run) {
        log.info("Processing tasks for runId: {}", run.getId());

        // Fresh step aggregator for this processing pass.
        // Cross-reference: .NET Run_Base.Process() — `_run_Task = new Run_Task(this)`.
        RunTask runTask = new RunTask(run, stepLimit);
        run.setRunTask(runTask);

        // Pass 1: Reset + Process each task
        for (TaskBase task : run.getTasks().values()) {
            if (!task.isTodo()) continue;
            if (task.isInError()) continue;
            try {
                // A task in degraded mode is NOT reset/processed; it is reported as a fail-soft error.
                // Cross-reference: .NET Process() — Errors.Add(Critique, TaskFailSoftMode) and skip.
                if (task.isModeDegrade()) {
                    run.getErrors().add(new RunError(CRITIQUE,
                            "Task " + task.getId() + " is in degraded mode (fail-soft) and was not processed"));
                    log.warn("Task {} is in degraded mode (fail-soft), not processed", task.getId());
                    continue;
                }
                log.debug("Processing task {}", task.getId());
                try {
                    executeTaskPhase(task::resetProcess,
                            "Error while resetting task " + task.getId() + " for " + run.getDebugInfo());
                } catch (EngineException controlled) {
                    recordTaskPhaseFailure(run, task, "reset", controlled);
                    continue;
                }
                try {
                    executeTaskPhase(() -> taskProcessService.process(task),
                            "Error while processing task " + task.getId() + " for " + run.getDebugInfo());
                } catch (EngineException controlled) {
                    recordTaskPhaseFailure(run, task, "process", controlled);
                    continue;
                }
                log.debug("Task {} produced {} blocsUpdate, {} blocsModify",
                        task.getId(), task.getBlocsUpdate().size(), task.getBlocsModify().size());
            } catch (Exception ex) {
                task.setInError(true);
                run.getErrors().add(new RunError(CRITIQUE, "Unable to execute task " + task.getId()));
                log.error("Task processing failed: task [{}], run [{}], cause [{}]",
                        task.getId(), run.getId(), ex.getClass().getSimpleName());
            }
        }

        // Pass 2: Post-process each task (e.g. DID, which needs all other tasks done first)
        for (TaskBase task : run.getTasks().values()) {
            if (!task.isTodo()) continue;
            if (task.isInError() || task.isModeDegrade()) continue;
            try {
                log.debug("Post-processing task {}", task.getId());
                executeTaskPhase(() -> taskPostProcessService.postProcess(task),
                        "Error while post-processing task " + task.getId() + " for " + run.getDebugInfo());
                log.debug("Task {} post-process: {} blocsUpdate, {} blocsModify",
                        task.getId(), task.getBlocsUpdate().size(), task.getBlocsModify().size());
            } catch (EngineException controlled) {
                task.setInError(true);
                run.getErrors().add(new RunError(CRITIQUE, controlled.getSafeMessageChain()));
                log.error("Task phase failed: phase [post-process], task [{}], run [{}], cause [{}]",
                        task.getId(), run.getId(), causeType(controlled));
            } catch (Exception ex) {
                task.setInError(true);
                run.getErrors().add(new RunError(CRITIQUE, "Unable to execute task " + task.getId()));
                log.error("Task post-processing failed: task [{}], run [{}], cause [{}]",
                        task.getId(), run.getId(), ex.getClass().getSimpleName());
            }
        }

        // Pass 3: Verify — a task with no blocs is an error; otherwise register its blocs
        for (TaskBase task : run.getTasks().values()) {
            if (!task.isTodo()) continue;
            if (task.isInError() || task.isModeDegrade()) continue;
            if (task.getBlocsUpdate().isEmpty() && task.getBlocsModify().isEmpty()) {
                // .NET Run_Base.Process pass 3: Errors.Add(TaskSansBloc, ...) → Unspecified (1), not Critique.
                run.getErrors().add(new RunError(RunError.UNSPECIFIED,
                        "Task " + task.getId() + " did not generate any block"));
                log.warn("Task {} has no blocs after processing", task.getDebugInfo());
            } else {
                run.getRunTask().addTask(task);
                log.debug("Task {} added to RunTask with {} blocsUpdate, {} blocsModify",
                        task.getId(), task.getBlocsUpdate().size(), task.getBlocsModify().size());
            }
        }

        log.info("Task processing completed for runId: {}", run.getId());
    }

        return cause != null ? cause.getClass().getSimpleName() : exception.getClass().getSimpleName();

                        decimalSignificative, oracleNumber, PERCENT_SUFFIX + PERCENT_SUFFIX, true);
            case DATE:
                return formatDate(value, nullString);
            case DATE_TIME:
                return formatDateTime(value, nullString);
            default:

                return text.isEmpty() ? nullString : text;

    private String formatInteger(Object value, boolean showZero, String nullString) {
        BigDecimal parsed = parseInvariantDecimal(value);
        if (parsed == null || parsed.compareTo(DECIMAL_UNSET) == 0) {
            return nullString;
        }
        int integer = CheckedDecimalArithmetic.toInt32Truncate(parsed);
        if (integer == Integer.MIN_VALUE || (integer == 0 && !showZero)) {
            return nullString;
        }
        return formatNumber(BigDecimal.valueOf(integer), 0, true);

    private String formatDecimal(Object value, int nbDecimal, boolean showZero, String nullString,

        boolean configuredPrecision = nbDecimal >= 0 && nbDecimal <= MAX_CONFIGURED_FRACTION_DIGITS;
        int fractionDigits = configuredPrecision ? nbDecimal : DEFAULT_FRACTION_DIGITS;
        boolean fixed = !configuredPrecision || decimalSignificative;
        return formatNumber(displayValue, fractionDigits, fixed) + suffix;

    private BigDecimal convertOracleNumber(BigDecimal value) {
        BigDecimal rounded = value.round(ORACLE_NUMBER_PRECISION);

            return null;
        }
        try {
            String normalized = normalizeInvariantNumber(raw);
            return CheckedDecimalArithmetic.roundToRepresentable(new BigDecimal(normalized));
        } catch (NumberFormatException | ArithmeticException ex) {
            return null;
        }

    private static String normalizeInvariantNumber(String raw) {
        String normalized = raw.trim().replace(" ", "").replace(",", "");
        boolean parenthesized = normalized.length() >= 2
                && normalized.charAt(0) == '(' && normalized.charAt(normalized.length() - 1) == ')';
        if (parenthesized) {
            normalized = normalized.substring(1, normalized.length() - 1);
        }
        if (normalized.endsWith("-") || normalized.endsWith("+")) {
            char trailingSign = normalized.charAt(normalized.length() - 1);
            normalized = normalized.substring(0, normalized.length() - 1);
            if (trailingSign == '-') normalized = "-" + normalized;
        }
        return parenthesized ? "-" + normalized : normalized;
    }

    }

    private static String indentLikeAcceptedWriter(String compactXml) {
        StringBuilder output = new StringBuilder(compactXml.length() + 128);
        int depth = 0;
        boolean previousWasText = false;
        boolean firstToken = true;

        for (String tag : tokenizeXml(compactXml)) {
            if (!tag.startsWith("<")) {
                output.append(tag);
                previousWasText = true;
                continue;
            }
            boolean closing = tag.startsWith("</");
            boolean selfClosing = tag.endsWith("/>");

            if (closing) depth--;
            if (!firstToken && !previousWasText) {
                output.append("\r\n").append("  ".repeat(depth));
            }
            output.append(selfClosing
                    ? tag.substring(0, tag.length() - 2) + " />"
                    : tag);
            if (!closing && !selfClosing) depth++;

            firstToken = false;
            previousWasText = false;
        }
        return output.toString();
    }

    private static List<String> tokenizeXml(String xml) {
        List<String> tokens = new ArrayList<>();
        int cursor = 0;
        while (cursor < xml.length()) {
            int tagStart = xml.indexOf('<', cursor);
            if (tagStart < 0) {
                tokens.add(xml.substring(cursor));
                break;
            }
            if (tagStart > cursor) tokens.add(xml.substring(cursor, tagStart));
            int tagEnd = findTagEnd(xml, tagStart + 1);
            tokens.add(xml.substring(tagStart, tagEnd + 1));
            cursor = tagEnd + 1;
        }
        return tokens;
    }

    private static int findTagEnd(String xml, int start) {
        char quote = 0;
        for (int index = start; index < xml.length(); index++) {
            char current = xml.charAt(index);
            if ((current == '\'' || current == '"')) {
                if (quote == 0) quote = current;
                else if (quote == current) quote = 0;
            } else if (current == '>' && quote == 0) {
                return index;
            }
        }
        throw new IllegalArgumentException("Unterminated XML tag");
    }

    private static boolean isOpeningTag(String token) {
        return token.startsWith("<") && !token.startsWith("</") && !token.endsWith("/>");
    }

    private static String tagName(String openingTag) {
        int end = openingTag.indexOf(' ');
        if (end < 0) end = openingTag.length() - 1;
        return openingTag.substring(1, end);
    }
